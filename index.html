// 🌟 修正版：自動抓取 Blogger 最新文章 (JSONP) function fetchLatestBloggerPost(blogUrl, titleId, dateId, snippetId, linkId) { const cleanUrl = blogUrl.replace(/\/$/, ''); // 隨機產生唯一的 Callback 函式名稱，防止快取與命名衝突 const cbName = 'cb_' + Math.random().toString(36).substring(2, 9); // 建立全域 Callback 函式 window[cbName] = function(data) { try { const entry = data.feed && data.feed.entry ? data.feed.entry[0] : null; if (!entry) { document.getElementById(titleId).innerText = "尚無公開文章"; document.getElementById(snippetId).innerText = "點擊右下角連結前往觀看..."; return; } // 1. 文章標題 const title = entry.title.$t; // 2. 發布日期 const published = new Date(entry.published.$t); const formattedDate = `${published.getFullYear()}/${published.getMonth() + 1}/${published.getDate()}`; // 3. 文章全文連結 let link = cleanUrl; if (entry.link) { const alternate = entry.link.find(l => l.rel === 'alternate'); if (alternate) link = alternate.href; } // 4. 文章摘要 (自動過濾 HTML 標籤) let rawContent = ''; if (entry.summary && entry.summary.$t) { rawContent = entry.summary.$t; } else if (entry.content && entry.content.$t) { rawContent = entry.content.$t; } // 清理 HTML 標籤並擷取前 80 個字 let cleanSnippet = rawContent.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ').trim(); if (cleanSnippet.length > 80) { cleanSnippet = cleanSnippet.substring(0, 80) + '...'; } // 更新 DOM 內容 const elTitle = document.getElementById(titleId); const elDate = document.getElementById(dateId); const elSnippet = document.getElementById(snippetId); const elLink = document.getElementById(linkId); if (elTitle) elTitle.innerText = title; if (elDate) elDate.innerText = formattedDate; if (elSnippet) elSnippet.innerText = cleanSnippet || '點擊閱讀完整文章...'; if (elLink) elLink.href = link; } catch (err) { console.error("解析 Blogger 文章失敗:", err); const elTitle = document.getElementById(titleId); if (elTitle) elTitle.innerText = "點擊前往閱讀文章"; } finally { // 清理暫存的全域 callback 函式 delete window[cbName]; } }; // 動態建立 script 標籤向 Blogger 請求 API const script = document.createElement('script'); script.src = `${cleanUrl}/feeds/posts/default?alt=json-in-script&max-results=1&callback=${cbName}`; script.onerror = function() { const elTitle = document.getElementById(titleId); const elSnippet = document.getElementById(snippetId); if (elTitle) elTitle.innerText = "無法載入文章"; if (elSnippet) elSnippet.innerText = "請點擊連結直接開啟 Blogger 部落格。"; }; document.head.appendChild(script); } function loadLatestBlogs() { fetchLatestBloggerPost('https://myezdailylife.blogspot.com', 'blog1-title', 'blog1-date', 'blog1-snippet', 'blog1-link'); fetchLatestBloggerPost('https://myeztravellife.blogspot.com', 'blog2-title', 'blog2-date', 'blog2-snippet', 'blog2-link'); }
