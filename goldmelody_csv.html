function rowsToSongs(rows) { if (!rows.length) return []; const headers = rows[0]; const titleCol = findColumn(headers, ["歌曲名稱", "歌曲", "歌名", "title", "song", "name"]); const viewsCol = findColumn(headers, ["熱門標籤", "views", "標籤", "分類", "描述"]); const coverCol = findColumn(headers, ["封面樣式", "封面圖片網址", "coverclass", "coverurl", "封面"]); const videoCol = findColumn(headers, ["youtube網址", "youtube連結", "影片網址", "影片連結", "video", "videoid", "youtubeid", "embedcode", "連結", "網址"]); const dataRows = titleCol >= 0 && videoCol >= 0 ? rows.slice(1) : rows; const titleIndex = titleCol >= 0 && videoCol >= 0 ? titleCol : 0; const videoIndex = titleCol >= 0 && videoCol >= 0 ? videoCol : 3; return dataRows.map(row => { const title = String(row[titleIndex] || "").trim(); const videoValue = String(row[videoIndex] || "").trim(); const videoId = extractYouTubeId(videoValue); if (!title || !videoId) return null; const coverValue = coverCol >= 0 ? String(row[coverCol] || "").trim() : ""; // 判斷 C 欄填的是「圖片網址」還是「CSS 代號」 let coverUrl = ""; let coverClass = ""; if (coverValue.startsWith("http://") || coverValue.startsWith("https://")) { coverUrl = coverValue; // 如果直接填網址，就用這張圖 } else { coverClass = coverValue; // 如果是英文代號，就對應原本的樣式 } const song = { title, views: viewsCol >= 0 && row[viewsCol] ? String(row[viewsCol]).trim() : "🎵 正在播放", isCustomCover: true, // 只要有填或有預設就開啟自訂封面 coverClass: coverClass, coverUrl: coverUrl, embedCode: `` }; return song; }).filter(Boolean); }
