Google Apps Script (GAS) 部署腳本

  1. 開啟 Google 雲端硬碟 ➔ 新增「Google Apps Script」專案。
  2. 複製下方腳本程式碼並覆蓋取代專案預設內容。
  3. 點選右上角「部署」➔「新增部署作業」➔ 選擇「網頁應用程式 (Web App)」。
  4. 將存取權限設為「所有人 (Anyone)」並部署獲得 URL。
function doGet(e) {
  var fileName = "myezlife_backup.json";
  var files = DriveApp.getFilesByName(fileName);
  if (files.hasNext()) {
    var content = files.next().getBlob().getDataAsString();
    return ContentService.createTextOutput(content).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({})).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var fileName = "myezlife_backup.json";
    var files = DriveApp.getFilesByName(fileName);
    var file = files.hasNext() ? files.next() : DriveApp.createFile(fileName, "");
    file.setContent(JSON.stringify(data, null, 2));
    return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
